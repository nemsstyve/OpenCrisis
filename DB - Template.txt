Table Directory [headercolor: #175e7a] {
	id integer [ pk, increment, not null ]
	First_name varchar
	Surname varchar
	Email varchar [ unique, note: 'Un seul email d''utilisateur dans toutes l''applications' ]
	Phone_number varchar [ unique ]
	Company varchar
	Platform_acces boolean
	Password char
	Preferences text
}

Table Role [headercolor: #175e7a] {
	id integer [ pk, increment, not null ]
	id_directory integer
}

Table Log_Book [headercolor: #175e7a] {
	id integer [ pk, increment, not null ]
	id_directory integer
	Timestamp timestamp
	Committee integer
	Message text
}

Table Timeline [headercolor: #175e7a] {
	id integer [ pk, increment, not null ]
	id_directory integer
	Event_timestamp timestamp
	Ip_address varchar
	Asset text
	Affected_user varchar
}

Ref fk_Role_id_directory_Directory {
	Role.id_directory > Directory.id [ delete: no action, update: no action ]
}

Ref fk_Log_Book_id_directory_Directory {
	Log_Book.id_directory > Directory.id [ delete: no action, update: no action ]
}

Ref fk_Timeline_id_directory_Directory {
	Timeline.id_directory > Directory.id [ delete: no action, update: no action ]
}
